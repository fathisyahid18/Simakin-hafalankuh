    <script>
        let students = JSON.parse(localStorage.getItem('hafalan_students')) || [];
        let currentScore = 100;
        let tempErrors = [];

        const juzStartPages = [
            1, 22, 42, 62, 82, 102, 122, 142, 162, 182, 
            202, 222, 242, 262, 282, 302, 322, 342, 362, 382, 
            402, 422, 442, 462, 482, 502, 522, 542, 562, 582
        ];

        const juzSelect = document.getElementById('juzSelect');
        const pageSelect = document.getElementById('pageSelect');
        
        for(let i=1; i<=30; i++) juzSelect.innerHTML += `<option value="${i}">Juz ${i}</option>`;

        function updatePageDropdown() {
            const juz = parseInt(juzSelect.value);
            let maxPages = 20;
            if (juz === 1) maxPages = 21;
            else if (juz === 30) maxPages = 23;

            pageSelect.innerHTML = '';
            for (let i = 1; i <= maxPages; i++) {
                pageSelect.innerHTML += `<option value="${i}">Halaman ${i}</option>`;
            }
            syncMushaf();
        }

        document.getElementById('studentForm').addEventListener('submit', function(e) {
            e.preventDefault();
            students.push({
                id: Date.now().toString(),
                name: document.getElementById('studentName').value.trim(),
                class: document.getElementById('studentClass').value.trim(),
                history: []
            });
            saveToStorage(); renderStudents(); this.reset();
        });

        function importExcel(e) {
            const file = e.target.files[0]; if (!file) return;
            const reader = new FileReader();
            reader.onload = function(evt) {
                try {
                    const data = evt.target.result;
                    const workbook = XLSX.read(data, { type: 'binary' });
                    const worksheet = workbook.Sheets[workbook.SheetNames[0]];
                    const json = XLSX.utils.sheet_to_json(worksheet, { header: 1 });
                    const timestamp = Date.now();
                    
                    for (let i = 1; i < json.length; i++) {
                        const row = json[i];
                        if (row && row[0]) {
                            students.push({
                                id: (timestamp + i).toString(),
                                name: row[0].toString().trim(),
                                class: row[1] ? row[1].toString().trim() : '-',
                                history: []
                            });
                        }
                    }
                    saveToStorage(); renderStudents(); alert('Sukses mengimpor data siswa!');
                } catch (err) { alert('Gagal membaca Excel.'); }
                e.target.value = '';
            };
            reader.readAsBinaryString(file);
        }

        function saveToStorage() { localStorage.setItem('hafalan_students', JSON.stringify(students)); }

        function renderStudents() {
            const select = document.getElementById('studentSelect');
            const tableBody = document.getElementById('studentTableBody');
            const currentSelected = select.value;
            
            select.innerHTML = '<option value="">-- Pilih Siswa --</option>';
            tableBody.innerHTML = '';

            students.forEach(s => {
                select.innerHTML += `<option value="${s.id}">${s.name} (${s.class})</option>`;
                tableBody.innerHTML += `
                    <tr class="border-b border-slate-100 text-xs sm:text-sm">
                        <td class="p-3 font-medium text-slate-900">${s.name}</td>
                        <td class="p-3 text-slate-500">${s.class}</td>
                        <td class="p-3"><span class="bg-emerald-50 text-emerald-700 px-2 py-0.5 rounded font-semibold">${s.history.length} Hal</span></td>
                        <td class="p-3"><button onclick="deleteStudent('${s.id}')" class="text-rose-500 hover:text-rose-700 font-semibold cursor-pointer">Hapus</button></td>
                    </tr>`;
            });
            select.value = currentSelected;
        }

        function loadStudentData() { resetScore(); syncMushaf(); }

        function deductScore(points, reason) {
            currentScore = Math.max(0, currentScore - points);
            document.getElementById('liveScore').innerText = currentScore;
            tempErrors.push(`${reason} (-${points})`);
            const logBox = document.getElementById('errorLog');
            if(tempErrors.length === 1) logBox.innerHTML = '';
            logBox.innerHTML += `<div>• ${reason} (-${points})</div>`;
        }

        function resetScore() {
            currentScore = 100; tempErrors = [];
            document.getElementById('liveScore').innerText = currentScore;
            document.getElementById('errorLog').innerHTML = '<span class="text-slate-400 italic">Belum ada kesalahan...</span>';
        }

        function syncMushaf() {
            const juz = parseInt(juzSelect.value);
            const pageInJuz = parseInt(pageSelect.value);
            const studentId = document.getElementById('studentSelect').value;
            const img = document.getElementById('mushafImage');
            const placeholder = document.getElementById('mushafPlaceholder');

            if (!studentId || !pageInJuz) {
                img.classList.add('hidden'); placeholder.classList.remove('hidden');
                document.getElementById('mushafInfo').innerText = "Halaman: -";
                return;
            }

            const globalPage = juzStartPages[juz - 1] + (pageInJuz - 1);
            
            // Menggunakan Server Image CDN Terbuka resmi milik Quran.com (Ramah File Lokal & Live Web)
            const targetUrl = `https://page.quran.com/v2/${globalPage}.png`;

            placeholder.innerHTML = `
                <div class="flex flex-col items-center justify-center p-4">
                    <div class="w-8 h-8 border-4 border-emerald-500 border-t-transparent rounded-full animate-spin mb-2"></div>
                    <p class="text-xs text-emerald-700 font-medium animate-pulse">Memuat halaman ${globalPage}...</p>
                </div>`;
            img.classList.add('hidden');
            placeholder.classList.remove('hidden');

            img.onload = function() { placeholder.classList.add('hidden'); img.classList.remove('hidden'); };
            img.onerror = function() { placeholder.innerHTML = `<p class="text-xs text-rose-500 font-semibold p-4">Gagal memuat halaman. Pastikan device terhubung internet.</p>`; };
            
            img.src = targetUrl;
            document.getElementById('mushafInfo').innerText = `Hal. ${globalPage} (Juz ${juz}, Syaf ke-${pageInJuz})`;
        }

        function saveSetoran() {
            const studentId = document.getElementById('studentSelect').value;
            if(!studentId) { alert('Pilih nama siswa dahulu!'); return; }

            const juz = parseInt(juzSelect.value);
            const page = parseInt(pageSelect.value);
            const student = students.find(s => s.id === studentId);
            const existingIndex = student.history.findIndex(h => h.juz == juz && h.page == page);
            
            const record = { juz, page, score: currentScore, errors: [...tempErrors], date: new Date().toLocaleDateString('id-ID') };

            if(existingIndex >= 0) {
                if(!confirm(`Sudah ada data Juz ${juz} Hal ${page}. Perbarui nilai?`)) return;
                student.history[existingIndex] = record;
            } else {
                student.history.push(record);
            }

            saveToStorage(); renderStudents(); resetScore();
            
            let maxPages = 20;
            if (juz === 1) maxPages = 21; else if (juz === 30) maxPages = 23;

            if(page < maxPages) { pageSelect.value = page + 1; } 
            else if(juz < 30) { juzSelect.value = juz + 1; updatePageDropdown(); }
            syncMushaf();
        }

        function deleteStudent(id) { if(confirm('Hapus siswa ini?')) { students = students.filter(s => s.id !== id); saveToStorage(); renderStudents(); loadStudentData(); } }
        function clearAllData() { if(confirm('Hapus semua data siswa?')) { students = []; saveToStorage(); renderStudents(); loadStudentData(); } }

        updatePageDropdown(); renderStudents();
    </script>
